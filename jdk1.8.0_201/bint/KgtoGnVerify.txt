import java.util.Scanner;
class KgtoGnVerify
{
 public static void main(String args [])
 {
  Scanner obj=new Scanner(System.in);
  System.out.print("Tell Your Kg Amount > ");
  int Kg=obj.nextInt();
  System.out.print("Here are your Grams > "+ (Kg*1000));
  System.out.println("To verify Tell Answer Here > ");
  int G=obj.nextInt();
  System.out.println(G/Kg);
 }
}
